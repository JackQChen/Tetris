# 修改root密码
sudo passwd root

# 允许root进行ssh登录
nano /etc/ssh/sshd_config
PermitRootLogin yes

# 树莓派换源
# 修改/etc/apt/sources.list内容将deb.debian.org域名换成
mirrors.aliyun.com
mirrors.tuna.tsinghua.edu.cn

# 安装.NET8
https://dotnet.microsoft.com/zh-cn/download/dotnet/8.0 选择ARM64版本
wget https://download.visualstudio.microsoft.com/download/dotnet-sdk-8.0.404-linux-arm64.tar.gz

# 解压文件内容到树莓派的/opt/dotnet文件夹中（创建dotnet文件夹），修改777权限
tar xvzf dotnet-sdk-8.0.404-linux-arm64.tar.gz

# 配置环境变量
nano ~/.bashrc
export DOTNET_ROOT=/opt/dotnet
export PATH=$PATH:/opt/dotnet

# 操作screen
sudo apt-get install screen
screen -S 1
screen -r 1
exit

# 修改音量
amixer -c 1 set 'PCM' 90%
alsamixer -> F6 -> bcm2835 Headphones -> 90

# 使用cvlc播放
# 在文件/usr/bin/vlc中将geteuid修改为getppid
cvlc --aout=alsa --alsa-audio-device=hw:1,0 --random --loop /root/Music/
